\section{Mệnh đề & Logic (Propositional Logic)}
% ===================================================

\subsection{Các Phép Toán Logic Cơ Bản}
Trong logic mệnh đề, chúng ta làm việc với các mệnh đề đúng ($T$) hoặc sai ($F$).

\begin{definitionbox}{Phép Kéo Theo (Implication)}
Cho hai mệnh đề $p$ và $q$. Mệnh đề kéo theo $p \rightarrow q$ chỉ sai khi $p$ đúng và $q$ sai.
\end{definitionbox}

Bảng chân trị cho phép kéo theo:
\begin{table}[h!]
\centering
\begin{tabular}{|c|c|c|}
\hline
\textbf{$p$} & \textbf{$q$} & \textbf{$p \rightarrow q$} \\ \hline
T & T & T \\ \hline
T & F & F \\ \hline
F & T & T \\ \hline
F & F & T \\ \hline
\end{tabular}
\caption{Bảng chân trị của $p \rightarrow q$}
\end{table}

\subsection{Thuật Toán Đường Đi Ngắn Nhất (Dijkstra)}
Thuật toán Dijkstra dùng để tìm đường đi ngắn nhất từ một đỉnh nguồn đến tất cả các đỉnh còn lại trong đồ thị có trọng số không âm.

\begin{notebox}{Điều kiện áp dụng}
Trọng số của tất cả các cạnh trong đồ thị phải \textbf{không âm} ($w(e) \ge 0$).
\end{notebox}

Ví dụ mã giả cho thuật toán Dijkstra:
\begin{lstlisting}[language=Python, caption=Mã giả Thuật toán Dijkstra]
import heapq

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    pq = [(0, start)]

    while pq:
        current_distance, current_node = heapq.heappop(pq)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(pq, (distance, neighbor))

    return distances
\end{lstlisting}

\subsection{Đại Số Boole}
\begin{itemize}
    \item \textbf{Luật De Morgan:} 
    \begin{equation}
        \overline{A \cdot B} = \bar{A} + \bar{B}
    \end{equation}
    \begin{equation}
        \overline{A + B} = \bar{A} \cdot \bar{B}
    \end{equation}
    \item \textbf{Luật Phân Phối:} 
    \begin{equation}
        A + (B \cdot C) = (A + B) \cdot (A + C)
    \end{equation}
\end{itemize}
